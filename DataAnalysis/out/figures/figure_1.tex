% =====================================================================
%  Helper: dotted, round-cornered self-loop denoting an "iterate until …"
%  step.  A repeat icon sits in a dotted circle (panel-coloured fill) on
%  the loop's horizontal edge — the circle outline matches the arrow's
%  line width.
%    #1 node   #2 anchor (south/north)   #3 colour
%    #4 panel fill   #5 signed depth in cm (e.g. -0.46 below, 0.46 above)
% =====================================================================
\newcommand{\reploop}[5]{%
  \coordinate (rlA)  at ([xshift=5mm]#1.#2);
  \coordinate (rlB)  at ([xshift=-5mm]#1.#2);
  \coordinate (rlA2) at ([yshift=#5cm]rlA);
  \coordinate (rlB2) at ([yshift=#5cm]rlB);
  \draw[#3, dotted, line width=0.9pt, rounded corners=2.5pt,
        -{Latex[length=1.4mm]}] (rlA) -- (rlA2) -- (rlB2) -- (rlB);
  \node[circle, draw=#3, dotted, line width=0.9pt, fill=#4,
        inner sep=0.4pt, minimum size=4.4mm]
        at ($(rlA2)!0.5!(rlB2)$) {\footnotesize\textcolor{#3}{\faSync}};
}

% =====================================================================
%  Helper: horizontal line with a colour gradient (#4 -> #5), used for
%  the result feeds entering the coordinators panel.
%    #1 x-start   #2 x-end   #3 y   #4 start colour   #5 end colour
% =====================================================================
\newcommand{\gradline}[5]{%
  \foreach \k in {0,...,23}{%
    \pgfmathsetmacro{\xa}{#1+(\k/24)*(#2-#1)}%
    \pgfmathsetmacro{\xb}{#1+((\k+1)/24)*(#2-#1)}%
    \pgfmathtruncatemacro{\pc}{100-(\k/23)*100}%
    \draw[line width=1.1pt, color=#4!\pc!#5] (\xa,#3) -- (\xb,#3);%
  }%
}
